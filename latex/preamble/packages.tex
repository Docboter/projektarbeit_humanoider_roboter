% preamble/packages.tex — Paketimporte

% ── Sprache & Zeichenkodierung ────────────────────────────────────────
\usepackage[ngerman]{babel}
\usepackage[autostyle=true, german=guillemets]{csquotes}   % dt. Anführungszeichen

% ── Schriften (LuaLaTeX + plex-otf) ──────────────────────────────────
% IBM Plex: moderne OpenType-Familie von IBM, in TeX Live 2021+ enthalten
\usepackage{plex-otf}                % setzt IBM Plex Serif / Sans / Mono automatisch

\usepackage{unicode-math}
\setmathfont{IBM Plex Math}          % passend zur IBM-Plex-Familie

  final,
]{microtype}

\usepackage[dvipsnames, svgnames, x11names]{xcolor}

\usepackage{graphicx}

\usepackage{float}              % [H]-Platzierung
\usepackage{wrapfig}            % Text umfließt Abbildung

\usepackage{minted}

\usepackage{amsmath}
\usepackage[
  locale              = DE,
  separate-uncertainty,
  per-mode            = symbol,
]{siunitx}

\usepackage{acro}

% ── Aufzählungen ─────────────────────────────────────────────────────

\usepackage[most]{tcolorbox}

% ── Diagramme & Plots ─────────────────────────────────────────────────
\usepackage{tikz}

  arrows.meta,
  positioning,
  shapes.geometric,
  fit,
  calc,
  decorations.pathreplacing,
  backgrounds,
}

\pgfplotsset{compat=1.18}

  urldate    = long,
  dashed     = false,
  maxnames   = 6,
  minnames   = 1,
]{biblatex}
