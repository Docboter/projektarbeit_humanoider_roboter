\chapter{SLURM-Batch-Skript (Auszug)}
\label{anh:slurm}

\begin{listing}[htb]
\begin{minted}{bash}
#!/usr/bin/env bash
#SBATCH --job-name=groot-finetune
#SBATCH --partition=kisski
#SBATCH --gres=gpu:a100:1
#SBATCH --time=48:00:00
#SBATCH --mem=64G
#SBATCH --cpus-per-task=8
#SBATCH --output=%x-%j.out
#SBATCH --error=%x-%j.err

module load apptainer

apptainer exec --nv \
  --bind /scratch/${USER}/data:/data \
  --env HF_TOKEN="${HF_TOKEN}" \
  --env WANDB_API_KEY="${WANDB_API_KEY:-}" \
  --env GLOBAL_BATCH_SIZE="${GLOBAL_BATCH_SIZE:-32}" \
  --env MAX_STEPS="${MAX_STEPS:-50000}" \
  "${SIF_IMAGE}" \
  bash /scripts/entrypoint.sh
\end{minted}
\caption{Vereinfachter Auszug aus \texttt{kisski\_submit.sh} (SLURM-Batch-Skript).}
\label{lst:anh-slurm}
\end{listing}
