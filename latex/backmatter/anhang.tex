\chapter{SLURM-Batch-Skript (Auszug)}
\label{anh:slurm}

\Cref{lst:anh-slurm} zeigt einen vereinfachten Auszug des für KISSKI verwendeten
Batch-Skripts. Das vollständige Skript liegt im Repository
(\doc{Training/kisski_submit.sh}), die Bedienung beschreibt
\doc{docs/training/kisski-hpc.md}.

\begin{listing}[htb]
\begin{minted}{bash}
#!/usr/bin/env bash
#SBATCH --job-name=groot-finetune
#SBATCH -p kisski                     # A100 80 GB (alt.: kisski-h100)
#SBATCH -N 1                          # alle GPUs auf einem Knoten (Lauf 75)
#SBATCH -G A100:4
#SBATCH -c 96
#SBATCH --mem=384G
#SBATCH -t 48:00:00
#SBATCH --output=logs/slurm-%j.out

module load apptainer

apptainer exec --nv \
  --bind /mnt/vast-kisski/projects/kisski-humrob/data:/data \
  --env HF_TOKEN="${HF_TOKEN}" \
  --env WANDB_API_KEY="${WANDB_API_KEY:-}" \
  --env GLOBAL_BATCH_SIZE="${GLOBAL_BATCH_SIZE:-32}" \
  --env MAX_STEPS="${MAX_STEPS:-44000}" \
  --env NUM_GPUS="${NUM_GPUS:-4}" \
  "${SIF_IMAGE}" \
  bash /scripts/entrypoint.sh
\end{minted}
\caption{Vereinfachter Auszug aus \texttt{kisski\_submit.sh}.}
\label{lst:anh-slurm}
\end{listing}
